cmake_minimum_required(VERSION 3.22.1)
project(openlock_jni C)
get_filename_component(OPENLOCK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../.." ABSOLUTE)
set(OPENLOCK_LIBRARY_DIR "${OPENLOCK_ROOT}/target/debug" CACHE PATH "Directory containing openlock_ffi")
add_library(openlock_jni SHARED openlock_jni.c)
target_include_directories(openlock_jni PRIVATE "${OPENLOCK_ROOT}/include")
if(NOT ANDROID)
    find_package(JNI REQUIRED)
    target_include_directories(openlock_jni PRIVATE ${JNI_INCLUDE_DIRS})
endif()
find_library(OPENLOCK_FFI_LIBRARY openlock_ffi PATHS "${OPENLOCK_LIBRARY_DIR}" REQUIRED NO_DEFAULT_PATH)
target_link_libraries(openlock_jni PRIVATE "${OPENLOCK_FFI_LIBRARY}")
set_target_properties(openlock_jni PROPERTIES BUILD_RPATH "${OPENLOCK_LIBRARY_DIR}")
target_compile_options(openlock_jni PRIVATE -Wall -Wextra -Werror)
